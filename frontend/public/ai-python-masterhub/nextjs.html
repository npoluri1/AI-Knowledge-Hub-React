1App Router Conventions: Layout, Page, Loading, Error

What & why. The App Router uses file-system conventions where layout.tsx wraps pages, page.tsx defines routes, loading.tsx shows Suspense fallbacks, and error.tsx catches runtime errors. This eliminates boilerplate routing configuration.

// app/layout.tsx — root layout wrapping all pages
export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <body>
        <nav>Navigation</nav>
        {children}
        <footer>Footer</footer>
      </body>
    </html>
  );
}

// app/page.tsx — home route at /
export default function HomePage() {
  return <h1>Welcome to Next.js</h1>;
}

// app/dashboard/loading.tsx — shown while dashboard loads
export default function DashboardLoading() {
  return <div className="skeleton">Loading dashboard...</div>;
}

// app/dashboard/error.tsx — catches errors in dashboard
'use client';
export default function DashboardError({ error, reset }: { error: Error; reset: () => void }) {
  return (
    <div>
      <p>Error: {error.message}</p>
      <button onClick={reset}>Try again</button>
    </div>
  );
}
Where you'll use itEvery Next.js app uses these conventions. Layouts persist across navigations, loading states provide instant feedback, and error boundaries isolate failures.
Under the hoodThe App Router uses React Server Components by default. Layouts are cached and only re-render when params change. loading.tsx wraps the nearest page in a Suspense boundary.

2Server-Side Rendering (SSR) Concept

What & why. SSR renders React components on the server, sending fully formed HTML to the browser. This improves initial load time, SEO, and works without JavaScript for the first paint. Next.js App Router components are server-rendered by default.

// Server Component (default) — rendered on the server
// Sends HTML, not JavaScript, to the browser
async function ProductPage({ params }) {
  const product = await fetchProduct(params.id);  // Direct DB/API call on server
  return (
    <div>
      <h1>{product.name}</h1>
      <p>{product.description}</p>
      <p>Price: ${product.price}</p>
    </div>
  );
}

// Dynamic route: SSR at request time (not build time)
// app/products/[id]/page.tsx
export default async function ProductPage({ params }) {
  const { id } = await params;
  const res = await fetch(`https://api.example.com/products/${id}`, {
    cache: 'no-store',  // Always fresh data = SSR
  });
  const product = await res.json();
  return <h1>{product.name}</h1>;
}
Where you'll use itSSR is used for pages with dynamic, user-specific content: dashboards, search results, and authenticated views that must be fresh on every request.
Under the hoodNext.js streams server-rendered HTML to the browser. The RSC payload is sent alongside HTML for hydration. Server Components send no JavaScript to the client, reducing bundle size.

3Static Site Generation (SSG) Concept

What & why. SSG pre-renders pages at build time, producing static HTML that can be served from a CDN with zero server computation. This gives the fastest possible load times for content that doesn't change per-request.

// Static page — built at build time, served from CDN
async function AboutPage() {
  const team = await fetchTeamMembers();  // Called at build time only
  return (
    <div>
      <h1>About Us</h1>
      {team.map(member => (
        <div key={member.id}>
          <h3>{member.name}</h3>
          <p>{member.role}</p>
        </div>
      ))}
    </div>
  );
}

// SSG with dynamic params via generateStaticParams
export async function generateStaticParams() {
  const posts = await db.post.findMany({ select: { slug: true } });
  return posts.map(p => ({ slug: p.slug }));
}

export default async function BlogPost({ params }) {
  const { slug } = await params;
  const post = await fetchPost(slug);
  return <article><h1>{post.title}</h1></article>;
}
Where you'll use itSSG is ideal for marketing pages, documentation, blog posts, and any content that can be pre-rendered and cached at the edge.
Under the hoodDuring build, Next.js calls the component, collects the HTML, and writes it to the .next/static directory. At runtime, the pre-built HTML is served directly without running any server-side code.

9Route Handlers (Simple API Routes)

What & why. Route handlers handle HTTP methods and return standard Response objects, supporting streaming, cookies, and headers.

// app/api/users/route.ts
import { NextRequest, NextResponse } from 'next/server';

export async function GET(request: NextRequest) {
  const searchParams = request.nextUrl.searchParams;
  const page = parseInt(searchParams.get('page') ?? '1');
  const users = await db.user.findMany({ skip: (page - 1) * 10, take: 10 });
  return NextResponse.json({ users, page });
}

export async function POST(request: NextRequest) {
  const body = await request.json();
  const user = await db.user.create({ data: body });
  return NextResponse.json(user, { status: 201 });
}

// app/api/users/[id]/route.ts
export async function GET(request, { params }: { params: { id: string } }) {
  const user = await db.user.findUnique({ where: { id: params.id } });
  if (!user) return NextResponse.json({ error: 'Not found' }, { status: 404 });
  return NextResponse.json(user);
}

// Streaming response
export async function GET() {
  const stream = new ReadableStream({
    async start(controller) {
      for (let i = 0; i < 10; i++) {
        controller.enqueue(new TextEncoder().encode(`data: ${i}\n\n`));
        await new Promise(r => setTimeout(r, 100));
      }
      controller.close();
    },
  });
  return new Response(stream, { headers: { 'Content-Type': 'text/event-stream' } });
}
Where you'll use itRoute handlers handle authentication, database queries, external API calls, and webhook processing.
Under the hoodRoute handlers run on Node.js by default (configurable to Edge). They use standard Web APIs (Request, Response).

10Dynamic Route Segments and Params

What & why. Dynamic routes use bracket syntax ([slug]) to capture URL segments. params are passed as async props to page and layout components in Next.js 15.

// app/products/[id]/page.tsx
interface Props {
  params: Promise<{ id: string }>;
  searchParams: Promise<{ color?: string; size?: string }>;
}

export default async function ProductPage({ params, searchParams }: Props) {
  const { id } = await params;
  const { color, size } = await searchParams;
  const product = await fetchProduct(id);
  return (
    <div>
      <h1>{product.name}</h1>
      {color && <p>Color: {color}</p>}
      {size && <p>Size: {size}</p>}
    </div>
  );
}

// Catch-all routes: app/docs/[...slug]/page.tsx
// /docs/a/b/c → params: { slug: ['a', 'b', 'c'] }

// Optional catch-all: app/shop/[[...slug]]/page.tsx
// /shop → params: { slug: undefined }
Where you'll use itProduct pages, blog posts, documentation, and user profiles all use dynamic routes.
Under the hoodIn Next.js 15, params and searchParams are Promises. The URL is parsed at request time and passed as structured props.

12Server vs Client Components (use-client)

What & why. Server Components run on the server and send only their rendered tree. Client Components ('use client') run in the browser and support interactivity. This split reduces client JavaScript.

// Server Component (default) — runs on the server
async function ProductList() {
  const products = await db.product.findMany();  // Direct DB access
  return (
    <ul>
      {products.map(p => <li key={p.id}>{p.name} — ${p.price}</li>)}
    </ul>
  );
}

// Client Component — runs in the browser
'use client';
import { useState } from 'react';

function AddToCartButton({ productId }: { productId: string }) {
  const [quantity, setQuantity] = useState(1);
  return (
    <div>
      <button onClick={() => setQuantity(q => Math.max(1, q - 1))}>-</button>
      <span>{quantity}</span>
      <button onClick={() => setQuantity(q => q + 1)}>+</button>
    </div>
  );
}

// Composition: server renders client component
export default async function ProductPage({ params }) {
  const product = await db.product.findUnique({ where: { id: params.id } });
  return (
    <div>
      <h1>{product.name}</h1>
      <AddToCartButton productId={product.id} />
    </div>
  );
}
Where you'll use itServer Components for data fetching and layout. Client Components for forms, modals, and interactive elements.
Under the hoodServer Components are serialized as RSC payload. Client Components receive the server tree plus their own hydration. Server-only code never reaches the client bundle.

13RSC Fetch Caching

What & why. Server Components cache fetch requests by default. Caching options control data freshness vs performance trade-offs.

// Default: cached until manually revalidated
async function CachedProducts() {
  const res = await fetch('https://api.example.com/products');
  return <ProductList data={await res.json()} />;
}

// No cache: always fresh
async function RealtimePrices() {
  const res = await fetch('https://api.example.com/prices', { cache: 'no-store' });
  return <PriceTicker data={await res.json()} />;
}

// Time-based revalidation
async function SemiFreshProducts() {
  const res = await fetch('https://api.example.com/products', {
    next: { revalidate: 60 },
  });
  return <ProductList data={await res.json()} />;
}

// Tag-based revalidation
async function TaggedProducts() {
  const res = await fetch('https://api.example.com/products', {
    next: { tags: ['products'] },
  });
  return <ProductList data={await res.json()} />;
}
Where you'll use itStatic content uses default caching. Dashboard data uses time-based. User data uses no-store. Product catalogs use tag-based.
Under the hoodNext.js maintains Data Cache (persistent) and Router Cache (in-memory). Fetch responses are stored with their cache key. Revalidation triggers background regeneration.

14Server Actions for Mutations

What & why. Server Actions are async functions running on the server, callable from any component. They replace manual API routes for form submissions and mutations.

// app/actions.ts
'use server';

import { revalidatePath } from 'next/cache';
import { redirect } from 'next/navigation';

export async function createPost(formData: FormData) {
  const title = formData.get('title') as string;
  const content = formData.get('content') as string;
  await db.post.create({ data: { title, content } });
  revalidatePath('/posts');
  redirect('/posts');
}

export async function deletePost(postId: string) {
  await db.post.delete({ where: { id: postId } });
  revalidatePath('/posts');
}

// Client Component calling Server Action
'use client';
function PostForm() {
  return (
    <form action={createPost}>
      <input name="title" required />
      <textarea name="content" required />
      <button type="submit">Create Post</button>
    </form>
  );
}
Where you'll use itServer Actions replace API routes for mutations: form submissions, CRUD operations, and state transitions.
Under the hoodServer Actions compile to HTTP POST endpoints. revalidatePath clears cached pages to reflect mutations.

16error.tsx and not-found.tsx

What & why. error.tsx catches runtime errors with a reset function. not-found.tsx handles 404 errors. Both isolate failures to their route segment.

// app/dashboard/error.tsx — must be a Client Component

export default function DashboardError({ error, reset }) {
  return (

      <h2>Something went wrong</h2>
      <p>{error.message}</p>
      {error.digest && <p>Digest: {error.digest}</p>}
      <button onClick={reset}>Try again</button>
    </div>
  );
}

// app/not-found.tsx — global 404 page
export default function NotFound() {
  return (
    <div>
      <h1>404 — Page Not Found</h1>
      <a href="/">Go Home</a>
    </div>
  );
}

// Triggering not-found
import { notFound } from 'next/navigation';
async function ProductPage({ params }) {
  const product = await db.product.findUnique({ where: { id: params.id } });

  return <h1>{product.name}</h1>;
}
Where you'll use iterror.tsx at each segment catches errors independently. not-found.tsx is triggered by notFound().

17Environment Variables: Server vs Public

What & why. Next.js distinguishes server-only env vars from public env vars (NEXT_PUBLIC_ prefix), preventing secrets from leaking to the browser.

# .env.local — server-only (NOT in bundle)
DATABASE_URL=postgresql://user:pass@localhost/db
API_SECRET_KEY=super-secret-key

# .env.local — public (NEXT_PUBLIC_ prefix)
NEXT_PUBLIC_API_URL=https://api.example.com

// Server Component (server-only)
async function Dashboard() {
  const db = new Database(process.env.DATABASE_URL);  // OK
  return <div>Data loaded</div>;
}

// Client Component (public only)
'use client';
function ApiStatus() {
  const apiUrl = process.env.NEXT_PUBLIC_API_URL;  // OK
  return <p>API: {apiUrl}</p>;
}

// NEVER in client: process.env.API_SECRET_KEY → undefined in browser
Where you'll use itEvery Next.js app uses env vars for database URLs, API keys, and configuration.
Under the hoodServer-only vars are inlined at build time for server bundles. NEXT_PUBLIC_ vars are inlined into client bundles. Others are stripped from client bundles.

18Middleware

What & why. Middleware runs before request completion, enabling auth checks, redirects, rewrites, and header manipulation.

// middleware.ts
import { NextRequest, NextResponse } from 'next/server';

export function middleware(request: NextRequest) {
  const token = request.cookies.get('auth-token');
  if (request.nextUrl.pathname.startsWith('/dashboard') && !token) {
    return NextResponse.redirect(new URL('/login', request.url));
  }
  const response = NextResponse.next();
  response.headers.set('x-request-id', crypto.randomUUID());
  return response;
}

export const config = {
  matcher: ['/dashboard/:path*', '/admin/:path*', '/api/:path*'],
};

// Rewrite pattern
export function middleware(request: NextRequest) {
  if (request.nextUrl.pathname.startsWith('/api/internal')) {
    return NextResponse.rewrite(
      new URL(request.nextUrl.pathname.replace('/api/internal', '/api'), request.url)
    );
  }
}
Where you'll use itAuth gating, A/B testing, geolocation redirects, and API proxying.
Under the hoodMiddleware runs on Edge Runtime by default (limited APIs). Keep it minimal for cold start performance.

19Four Cache Layers: Data, Full Route, Router, Memoization

What & why. Next.js has four cache layers, each serving a different purpose. Understanding when each applies prevents stale data bugs.

// 1. Data Cache — persists fetch responses across deployments
fetch(url, { cache: 'force-cache' })      // Cache forever
fetch(url, { cache: 'no-store' })          // Never cache
fetch(url, { next: { revalidate: 3600 } }) // Revalidate hourly

// 2. Full Route Cache — caches rendered HTML/RSC at build time
export const dynamic = 'force-dynamic';  // Disable full route cache

// 3. Router Cache — client-side cache for navigations
// In-memory, per-session, 30 seconds for static pages
// Clear with router.refresh()

// 4. Memoization — within a single render pass
// React memoizes fetch calls with same URL+options
Where you'll use itData Cache for API responses. Full Route Cache for static pages. Router Cache for navigation speed.
Under the hoodData Cache is on disk. Full Route Cache is pre-rendered HTML. Router Cache is a browser Map. Memoization is per-render.

20revalidatePath and revalidateTag

What & why. On-demand revalidation invalidates cached pages immediately after mutations, rather than waiting for time-based revalidation.

// app/actions.ts
'use server';
import { revalidatePath, revalidateTag } from 'next/cache';

export async function updateProduct(id: string, data: FormData) {
  await db.product.update({ where: { id }, data: { name: data.get('name') } });
  revalidatePath(`/products/${id}`);
  revalidateTag('products');
}

export async function createPost(data: FormData) {
  await db.post.create({ data: { title: data.get('title') } });
  revalidatePath('/blog');
  revalidatePath('/blog', 'layout');
}

// Usage with fetch tags
async function ProductPage({ params }) {
  const res = await fetch(`/api/products/${params.id}`, {
    next: { tags: [`product-${params.id}`] },
  });
  return <ProductDetail data={await res.json()} />;
}
Where you'll use itAfter any database mutation, call revalidatePath or revalidateTag.
Under the hoodrevalidatePath marks routes for regeneration on next request. revalidateTag marks all tagged fetches for cache invalidation.

21ISR (Incremental Static Regeneration)

What & why. ISR combines static generation with dynamic data. Pages are pre-rendered and revalidated on-demand or at intervals.

// Time-based ISR: revalidate every 60 seconds
async function ProductPage({ params }) {
  const res = await fetch(`/api/products/${params.id}`, {
    next: { revalidate: 60 },
  });
  return <ProductDetail data={await res.json()} />;
}

// On-demand ISR via revalidateTag
async function BlogPost({ params }) {
  const res = await fetch(`/api/posts/${params.slug}`, {
    next: { tags: ['blog'] },
  });
  return <article>{/* post */}</article>;
}

// Trigger from Server Action
'use server';
import { revalidateTag } from 'next/cache';
export async function publishPost(postId: string) {
  await db.post.update({ where: { id: postId }, data: { published: true } });
  revalidateTag('blog');
}

// app/api/chat/route.ts
import OpenAI from 'openai';
const openai = new OpenAI();

export async function POST(request: Request) {
  const { messages, model = 'gpt-4' } = await request.json();
  const stream = await openai.chat.completions.create({ model, messages, stream: true });

  const encoder = new TextEncoder();
  const readable = new ReadableStream({
    async start(controller) {
      for await (const chunk of stream) {
        const token = chunk.choices[0]?.delta?.content ?? '';
        if (token) controller.enqueue(encoder.encode(`0:${JSON.stringify(token)}\n`));
      }
      controller.close();
    },
  });

  return new Response(readable, { headers: { 'Content-Type': 'text/plain; charset=utf-8' } });
}
