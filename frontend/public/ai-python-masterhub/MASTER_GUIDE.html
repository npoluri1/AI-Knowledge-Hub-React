"""One-file tour of daily-driver Python: typing, dataclasses, decorators,
generators, context managers, pathlib, logging, walrus, match, asyncio."""
from __future__ import annotations
import asyncio, logging, time
from dataclasses import dataclass, field
from functools import lru_cache, wraps
from pathlib import Path
from typing import Iterable, Iterator

log = logging.getLogger("tour")

# ---------- dataclasses + typing ----------
@dataclass(frozen=True, order=True)
class Task:
    priority: int                      # order=True -> tasks sortable by priority
    title: str
    tags: list[str] = field(default_factory=list)

    def is_urgent(self) -> bool:
        return self.priority <= 2

# ---------- decorator with timing ----------
def timed(fn):
    @wraps(fn)
    def inner(*args, **kwargs):
        t0 = time.perf_counter()
        try:
            return fn(*args, **kwargs)
        finally:
            log.info("%s took %.3fs", fn.__name__, time.perf_counter() - t0)
    return inner

# ---------- generator pipeline (lazy) ----------
def read_numbers(path: Path) -> Iterator[int]:
    for line in path.read_text().splitlines():        # pretend huge file
        if line.strip().isdigit():
            yield int(line)

@timed
def sum_big(path: Path) -> int:
    return sum(n for n in read_numbers(path) if n > 10)   # nothing materialized

# ---------- memoized pure function ----------
@lru_cache(maxsize=None)
def fib(n: int) -> int:
    return n if n < 2 else fib(n - 1) + fib(n - 2)

# ---------- context manager ----------
from contextlib import contextmanager

@contextmanager
def stage(name: str):
    log.info("-> %s", name)
    try:
        yield
    except Exception:
        log.exception("stage failed: %s", name)
        raise
    else:
        log.info("<- %s ok", name)

# ---------- walrus + match ----------
def classify(tasks: Iterable[Task]) -> dict[str, list[str]]:
    buckets: dict[str, list[str]] = {}
    for t in tasks:
        match t:
            case Task(priority=1, title=title): buckets.setdefault("p1", []).append(title)
            case Task(priority=p) if p <= 3:    buckets.setdefault("mid", []).append(t.title)
            case _:                             buckets.setdefault("low", []).append(t.title)
    return buckets

# ---------- asyncio ----------
async def fetch(i: int) -> int:
    await asyncio.sleep(0.1 * i)          # simulated IO
    return i * i

async def main() -> None:
    results = await asyncio.gather(*(fetch(i) for i in range(5)))
    print("gathered:", results)

if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
    tmp = Path("nums.txt"); tmp.write_text("\n".join(["5", "20", "3", "99"]))
    with stage("sum"):
        print("sum_big:", sum_big(tmp))
    with stage("fib"):
        print("fib(90) =", fib(90))
    with stage("classify"):
        ts = [Task(1, "hotfix"), Task(2, "deploy", ["infra"]), Task(9, "docs")]
        print(classify(ts))
    asyncio.run(main())

from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass
from enum import Enum, auto
from typing import Protocol, runtime_checkable

class Shape(ABC):
    @abstractmethod
    def area(self) -> float: ...
    def describe(self) -> str:            # concrete method on abstract class
        return f"{type(self).__name__} area={self.area():.2f}"

@dataclass(frozen=True, slots=True)
class Circle(Shape):
    r: float
    def area(self) -> float: return 3.14159 * self.r ** 2

@runtime_checkable
class Drawable(Protocol):                 # static duck typing
    def area(self) -> float: ...

class Color(Enum):
    RED = auto(); GREEN = auto()

class Temperature:
    def __init__(self, celsius: float = 0): self._c = celsius
    @property
    def celsius(self) -> float: return self._c
    @celsius.setter
    def celsius(self, v: float):
        if v < -273.15: raise ValueError("below absolute zero")
        self._c = v
    @property
    def fahrenheit(self) -> float: return self._c * 9 / 5 + 32

class Vector:
    __slots__ = ("x", "y")
    def __init__(self, x: float, y: float): self.x, self.y = x, y
    def __add__(self, o: "Vector") -> "Vector": return Vector(self.x + o.x, self.y + o.y)
    def __repr__(self) -> str: return f"Vector({self.x}, {self.y})"
    def __eq__(self, o: object) -> bool:
        return isinstance(o, Vector) and (self.x, self.y) == (o.x, o.y)
    def __hash__(self) -> int: return hash((self.x, self.y))

if __name__ == "__main__":
    shapes: list[Shape] = [Circle(1), Circle(2.5)]
    for s in shapes:
        assert isinstance(s, Drawable)     # protocol runtime check
        print(s.describe())
    t = Temperature(); t.celsius = 100
    print("100C =", t.fahrenheit, "F")
    print(Vector(1, 2) + Vector(3, 4), Vector(1, 1) == Vector(1, 1))
    try:
        t.celsius = -300
    except ValueError as e:
        print("blocked:", e)

import asyncio, random, time

async def producer(q: asyncio.Queue, n: int):
    for i in range(n):
        await asyncio.sleep(random.uniform(0.01, 0.05))   # simulate IO latency
        await q.put({"job": i})
        print(f"[producer] queued job={i} size={q.qsize()}")
    for _ in range(WORKERS):
        await q.put(None)                                  # poison pills

async def worker(q: asyncio.Queue, sem: asyncio.Semaphore, wid: int):
    while True:
        item = await q.get()
        if item is None:
            q.task_done(); break
        async with sem:                                    # cap concurrency at 2
            await asyncio.sleep(random.uniform(0.05, 0.15))
            print(f"[worker{wid}] done job={item['job']}")
        q.task_done()

WORKERS = 3
async def main():
    q: asyncio.Queue = asyncio.Queue(maxsize=4)            # bounded => backpressure
    sem = asyncio.Semaphore(2)
    prod = asyncio.create_task(producer(q, 8))
    consumers = [asyncio.create_task(worker(q, sem, w)) for w in range(WORKERS)]
    await prod; await q.join()
    await asyncio.gather(*consumers)
    print("pipeline drained")

if __name__ == "__main__":
    t0 = time.perf_counter(); asyncio.run(main())
    print(f"elapsed {time.perf_counter()-t0:.2f}s")

import pytest

# --- code under test ---
def discount(price: float, pct: float) -> float:
    if price < 0:      raise ValueError("negative price")
    if not 0 <= pct <= 90: raise ValueError("pct out of range")
    return round(price * (1 - pct / 100), 2)

class PaymentGateway:                       # external dependency (fake)
    def charge(self, cents: int) -> bool: raise RuntimeError("network!")

def checkout(gw: PaymentGateway, price: float) -> bool:
    # float trap: int(19.99*100) == 1998!  always round before int()
    return gw.charge(int(round(price * 100)))

# --- tests ---
@pytest.mark.parametrize("price,pct,expected", [
    (100, 10, 90.0), (199.99, 50, 100.0), (80, 0, 80.0),
])
def test_discount_happy(price, pct, expected):
    assert discount(price, pct) == expected

@pytest.mark.parametrize("price,pct", [(-1, 10), (100, 95), (100, -5)])
def test_discount_rejects(price, pct):

import csv, sqlite3, statistics, tempfile, pathlib

RAW = """id,name,dept,salary
1,Ada,eng,180000
2,Grace,eng,195000
3,Alan,math,
4,Katherine,nasa,150000
bad,row,x
5,Edsger,math,175000"""

def extract(raw: str):
    for row in csv.DictReader(raw.splitlines()):
        try:
            yield {"id": int(row["id"]), "name": row["name"].strip(),
                   "dept": row["dept"].lower(), "salary": float(row["salary"])}
        except (KeyError, ValueError) as e:
            print("skipped bad row:", row, "->", e)

def load(rows, db: pathlib.Path):
    con = sqlite3.connect(db); con.row_factory = sqlite3.Row
    con.execute("CREATE TABLE IF NOT EXISTS emp(id INTEGER PRIMARY KEY, name TEXT, dept TEXT, salary REAL)")
    with con:
        con.executemany("INSERT INTO emp VALUES(:id,:name,:dept,:salary)",
                        [r for r in rows if r.get("salary")])
    return con

if __name__ == "__main__":
    db = pathlib.Path(tempfile.gettempdir()) / "hr.db"
    db.unlink(missing_ok=True)
    con = load(extract(RAW), db)
    rows = con.execute("SELECT dept, salary FROM emp").fetchall()
    by_dept: dict[str, list[float]] = {}
    for r in rows: by_dept.setdefault(r["dept"], []).append(r["salary"])
    for dept, pays in sorted(by_dept.items()):
        print(f"{dept:6} n={len(pays)} mean={statistics.mean(pays):,.0f} max={max(pays):,.0f}")
    con.close()

dev = "cuda" if torch.cuda.is_available() else "cpu"

def make_blobs(n=4096, k=3, d=16):
    centers = torch.randn(k, d) * 4
    y = torch.randint(0, k, (n,))
    X = centers[y] + 0.9 * torch.randn(n, d)
    return X.to(dev), y.to(dev)

class MLP(nn.Sequential):
    def __init__(self, d_in, h, classes):
        super().__init__(
            nn.Linear(d_in, h), nn.ReLU(), nn.BatchNorm1d(h), nn.Dropout(0.1),
            nn.Linear(h, h), nn.GELU(), nn.LayerNorm(h),
            nn.Linear(h, classes))

X, y = make_blobs()
n_tr = int(0.8 * len(X))
Xtr, Xva, ytr, yva = X[:n_tr], X[n_tr:], y[:n_tr], y[n_tr:]

opt = torch.optim.AdamW(model.parameters(), lr=3e-3, weight_decay=1e-2)

best_acc, patience, bad = 0.0, 5, 0
for epoch in range(30):
    model.train(); perm = torch.randperm(n_tr)
    for i in range(0, n_tr, 128):
        idx = perm[i:i+128]
        opt.zero_grad(set_to_none=True)
        loss = lossf(model(Xtr[idx]), ytr[idx])

import numpy as np

DIM = 256
def embed_offline(text: str) -> np.ndarray:
    v = np.zeros(DIM, dtype=np.float32)
    for tok in text.lower().split():
        v[int(hashlib.md5(tok.encode()).hexdigest(), 16) % DIM] += 1.0
    n = np.linalg.norm(v)
    return v / n if n else v

    from sentence_transformers import SentenceTransformer
    _st = SentenceTransformer("all-MiniLM-L6-v2")
    def embed(text): return _st.encode(text, normalize_embeddings=True)
except Exception:
    embed = embed_offline

class MiniVectorStore:
    def __init__(self): self.ids, self.docs, self.M = [], [], None
    def add(self, ids, docs):
        V = np.stack([embed(d) for d in docs]).astype(np.float32)
        self.ids += ids; self.docs += docs
        self.M = V if self.M is None else np.vstack([self.M, V])
    def search(self, q, k=3):
        sims = self.M @ embed(q)                        # normalized => dot == cosine
        top = np.argsort(-sims)[:k]
        return [(self.ids[i], float(sims[i]), self.docs[i]) for i in top]
    def save(self, p): pathlib.Path(p).write_text(json.dumps(
        {"ids": self.ids, "docs": self.docs, "M": self.M.tolist()}))

import torch, torch.nn as nn, copy

base.weight.requires_grad_(False)                  # FREEZE backbone

class LoRALinear(nn.Module):                       # wrap frozen layer
    def __init__(self, wrapped: nn.Linear, r=4, alpha=8.0):
        super().__init__()
        self.wrapped = wrapped
        self.A = nn.Parameter(torch.randn(r, wrapped.in_features) * 0.01)
        self.B = nn.Parameter(torch.zeros(wrapped.out_features, r))   # B=0 start ⇒ ΔW=0
        self.scale = alpha / r
    def forward(self, x):
        return self.wrapped(x) + self.scale * (x @ self.A.T @ self.B.T)

model = LoRALinear(base, rank, alpha)
print("trainable:", sum(p.numel() for p in model.parameters() if p.requires_grad),
      "| frozen:", base.weight.numel())

import numpy as np

rng = np.random.default_rng(0)
W = rng.normal(size=(64, 32)).astype(np.float32) * np.linspace(0.2, 1.0, 32)

def quant_per_tensor(W, bits=8):
    qmax = 2 ** (bits - 1) - 1
    s = np.abs(W).max() / qmax                       # one scale for whole tensor
    Q = np.clip(np.round(W / s), -qmax - 1, qmax).astype(np.int8)
    return Q, s

def quant_per_channel(W, axis=1, bits=8):            # one scale per output channel
    qmax = 2 ** (bits - 1) - 1
    s = np.abs(W).max(axis=axis, keepdims=True) / qmax
    Q = np.clip(np.round(W / s), -qmax - 1, qmax).astype(np.int8)
    return Q, s

def dequant(Q, s): return Q.astype(np.float32) * s

def mse(a, b): return ((a - b) ** 2).mean()

Qt, st = quant_per_tensor(W)
Qc, sc = quant_per_channel(W)
print(f"fp32 bytes      : {W.nbytes}")
print(f"int8 bytes      : {Qt.nbytes}  ({W.nbytes/Qt.nbytes:.1f}x smaller)")
print(f"MSE per-tensor  : {mse(W, dequant(Qt, st)):.6f}")
print(f"MSE per-channel : {mse(W, dequant(Qc, sc)):.6f}   <- usually better")
print("memory saved    : %.1f%%" % (100 * (1 - Qt.nbytes / W.nbytes)))

def psi(expected, actual, bins=10):
    edges = np.percentile(expected, np.linspace(0, 100, bins + 1)); edges[0], edges[-1] = -np.inf, np.inf
    e = np.histogram(expected, edges)[0] / len(expected)
    a = np.histogram(actual, edges)[0] / max(len(actual), 1)
    e, a = np.clip(e, 1e-6, None), np.clip(a, 1e-6, None)
    return float(((a - e) * np.log(a / e)).sum())

@register("book_meeting")

    datetime.date.fromisoformat(date)                       # validate

    return {"city": city, "temp_c": 27, "sky": "partly cloudy"}  # stub

def validate(args: dict, schema: dict):
    props, req = schema["properties"], schema.get("required", [])
    missing = [r for r in req if r not in args]
    if missing: raise ValueError(f"missing {missing}")
    out = {}
    for k, v in args.items():
        t = props[k]["type"]
        ok = {"string": str, "integer": int}[t]             # minimal demo
        if not isinstance(v, ok): raise ValueError(f"{k} must be {t}")
        if "enum" in props[k] and v not in props[k]["enum"]:
            raise ValueError(f"{k} must be one of {props[k]['enum']}")
        out[k] = v
    return out

from collections import deque
import numpy as np, hashlib

DIM = 128
def emb(text):
    v = np.zeros(DIM); [v.__setitem__(int(hashlib.md5(t.encode()).hexdigest(),16)%DIM,
        v[int(hashlib.md5(t.encode()).hexdigest(),16)%DIM]+1) for t in text.lower().split()]
    n = np.linalg.norm(v); return v/n if n else v

class ShortTerm:                                  # sliding window
    def __init__(self, k=6): self.buf = deque(maxlen=k)
    def add(self, role, text): self.buf.append((role, text))
    def render(self): return "\n".join(f"{r}: {t}" for r, t in self.buf)

class SummaryMemory:                              # rolling compression
    def __init__(self, trigger=6): self.st = ShortTerm(trigger); self.digest = "(none)"
    def add(self, role, text):
        self.st.add(role, text)
        if len(self.st.buf) == self.st.buf.maxlen:
            old = "\n".join(t for _, t in list(self.st.buf)[:3])
            self.digest = (self.digest + " | " +
                " ".join(old.split()[:25]))[-400:]     # truncate-summarize heuristic
            for _ in range(3): self.st.buf.popleft()
    def render(self): return f"EARLIER: {self.digest}\nRECENT:\n{self.st.render()}"

class LongTerm:                                   # episodic vector store
    def __init__(self): self.V, self.docs = None, []
    def remember(self, text):
        self.docs.append(text)
        e = emb(text)[None, :]
        self.V = e if self.V is None else np.vstack([self.V, e])
    def recall(self, q, k=2):
        if self.V is None: return []
        sims = self.V @ emb(q); idx = np.argsort(-sims)[:k]
        return [(self.docs[i], float(sims[i])) for i in idx]

class MemoryManager:
    def __init__(self): self.work = SummaryMemory(); self.lt = LongTerm()
    def turn(self, user_text):
        hits = self.lt.recall(user_text)
        ctx = self.work.render() + ("\nRELEVANT PAST:\n" +
              "\n".join(d for d, _ in hits) if hits else "")
        self.work.add("user", user_text); self.lt.remember(user_text)
        reply = f"(echo) {user_text}"               # replace with your LLM
        self.work.add("assistant", reply)
        return ctx, reply

if __name__ == "__main__":
    m = MemoryManager()
    for u in ["I prefer dark mode.", "My project codename is BLUEFIN.",
              "What was my project codename?", "Remind me my UI preference?"]:
        ctx, reply = m.turn(u)
        print("=" * 60); print(ctx[:400]); print("--> ", reply)

def call_tool(name, args):
    if name == "now":
        return datetime.datetime.utcnow().isoformat() + "Z"
    if name == "sha256":
        return hashlib.sha256(args["text"].encode()).hexdigest()
    raise ValueError(f"unknown tool {name}")

def handle(req):
    method = req.get("method", "")
    rid = req.get("id")
    def ok(result): return {"jsonrpc": "2.0", "id": rid, "result": result}
    def err(code, msg): return {"jsonrpc": "2.0", "id": rid,
                                "error": {"code": code, "message": msg}}
    if method == "initialize":
        return ok({"protocolVersion": PROTOCOL, "capabilities": {"tools": {}},
                   "serverInfo": SERVER})
    if method == "tools/list":
        return ok({"tools": TOOLS_SPEC})
    if method == "tools/call":
        p = req.get("params", {})
        try:
            text = str(call_tool(p["name"], p.get("arguments", {})))
            return ok({"content": [{"type": "text", "text": text}], "isError": False})
        except Exception as e:
            return ok({"content": [{"type": "text", "text": str(e)}], "isError": True})
    if method.startswith("notifications/"):
        return None                                        # no response for notifications
    return err(-32601, f"method not found: {method}")

import sys, time, json, uuid, contextlib
sys.stdout.reconfigure(encoding="utf-8", errors="replace")   # Windows cp1252-safe

class Tracer:
    def __init__(self): self.spans, self.stack = [], []
    @contextlib.contextmanager
    def span(self, name, **attrs):
        sid, parent = uuid.uuid4().hex[:8], (self.stack[-1] if self.stack else None)
        rec = {"id": sid, "parent": parent, "name": name, "attrs": attrs,
               "t0": time.time(), "children": []}
        self.spans.append(rec)
        if parent:
            next(s for s in self.spans if s["id"] == parent)["children"].append(sid)
        self.stack.append(sid)
        try:
            yield rec
        except Exception as e:
            rec["attrs"]["error"] = f"{type(e).__name__}: {e}"; raise
        finally:
            rec["dt_ms"] = round((time.time() - rec["t0"]) * 1000, 2)
            self.stack.pop()

    def render(self, sid=None, depth=0):
        roots = [s for s in self.spans if s["parent"] == (sid if sid else None)]
        for s in roots:
            err = " !!ERROR" if "error" in s["attrs"] else ""
            print("  " * depth + f"└─ {s['name']} ({s['dt_ms']}ms){err} {dict(list(s['attrs'].items())[:2])}")
            self.render(s["id"], depth + 1)

    hdr = b64(json.dumps({"alg": "HS256", "typ": "JWT"}).encode())

  const [v, setV] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => setV(value), ms);
    return () => clearTimeout(t);            // cleanup cancels stale timers
  }, [value, ms]);
  return v;
}

import random, sys

W, H = 800, 500
pygame.init()
screen = pygame.display.set_mode((W, H))
clock = pygame.time.Clock()
FPS = 60

class Ball:
    def __init__(self):
        self.r = random.randint(12, 26)
        self.x, self.y = random.randint(self.r, W-self.r), random.randint(self.r, H-self.r)
        ang = random.uniform(0, 6.283)
        speed = random.uniform(120, 260)              # pixels per SECOND
        self.vx, self.vy = speed*random.cos(ang), speed*random.sin(ang)
        self.color = pygame.Color(random.randint(60,255),
                                  random.randint(60,255), random.randint(60,255))
    def step(self, dt: float):                        # dt = seconds since last frame
        self.x += self.vx * dt; self.y += self.vy * dt
        if not self.r <= self.x <= W-self.r: self.vx *= -1
        if not self.r <= self.y <= H-self.r: self.vy *= -1
    def draw(self, surf):
        pygame.draw.circle(surf, self.color, (int(self.x), int(self.y)), self.r)

balls = [Ball() for _ in range(14)]
while True:
    dt = clock.tick(FPS) / 1000.0                     # frame-time in seconds
    for e in pygame.event.get():
        if e.type == pygame.QUIT or (
           e.type == pygame.KEYDOWN and e.key == pygame.K_ESCAPE): sys.exit()
    screen.fill((15, 18, 32))
    for b in balls:
        b.step(dt); b.draw(screen)                    # physics scaled by real dt
    pygame.display.flip()
