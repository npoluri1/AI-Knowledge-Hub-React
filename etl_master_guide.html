WHAT IS KAFKA?

CORE CONCEPTS

Architecture

  • Topic: Logical channel for messages
  • Partition: Parallelism unit within topic

  • KRaft: New coordination (no ZooKeeper)

Kafka Connect & Streams

  • Single Message Transform (SMT)

# Producer (confluent-kafka)
from confluent_kafka import Producer
import json

producer = Producer({'bootstrap.servers': 'localhost:9092'})

def delivery_report(err, msg):
    if err:
        print(f"Delivery failed: {err}")

producer.produce(
    topic='orders',
    key='order_001',
    value=json.dumps({
        'order_id': 'order_001',
        'customer_id': 'cust_123',
        'amount': 7500.00,
        'region': 'APAC'
    }),
    callback=delivery_report
)
producer.flush()

# Consumer (confluent-kafka)
from confluent_kafka import Consumer
import json

consumer = Consumer({
    'bootstrap.servers': 'localhost:9092',
    'group.id': 'order-processor',
    'auto.offset.reset': 'earliest'
})
consumer.subscribe(['orders'])

while True:
    msg = consumer.poll(1.0)
    if msg is None: continue
    if msg.error(): continue
    data = json.loads(msg.value().decode('utf-8'))
    print(f"Processing order: {data['order_id']}")
    consumer.commit(msg)

SQL/QUERY EXAMPLES (Flink SQL)

-- Create Kafka source table
CREATE TABLE orders (
  order_id STRING,
  customer_id STRING,
  amount DECIMAL(10,2),
  region STRING,
  order_time TIMESTAMP(3),
  WATERMARK FOR order_time AS order_time - INTERVAL '5' SECOND
) WITH (
  'connector' = 'kafka',
  'topic' = 'orders',
  'properties.bootstrap.servers' = 'localhost:9092',
  'format' = 'avro'
);

-- Windowed aggregation: Revenue per region per 5 minutes
SELECT
  region,
  TUMBLE_START(order_time, INTERVAL '5' MINUTE) AS window_start,
  COUNT(*) AS order_count,
  SUM(amount) AS total_revenue
FROM orders
GROUP BY region, TUMBLE(order_time, INTERVAL '5' MINUTE);

-- Session window: Group by customer activity
SELECT
  customer_id,
  SESSION_START(order_time, INTERVAL '30' MINUTE) AS session_start,
  SESSION_END(order_time, INTERVAL '30' MINUTE) AS session_end,
  COUNT(*) AS order_count,
  SUM(amount) AS session_total
FROM orders
GROUP BY customer_id, SESSION(order_time, INTERVAL '30' MINUTE);

-- Top-N: Top 5 regions by revenue per hour
SELECT * FROM (
  SELECT
    region,
    TUMBLE_START(order_time, INTERVAL '1' HOUR) AS window_start,
    SUM(amount) AS total_revenue,
    ROW_NUMBER() OVER (
      PARTITION BY TUMBLE_START(order_time, INTERVAL '1' HOUR)
      ORDER BY SUM(amount) DESC
    ) AS row_num
  FROM orders
  GROUP BY region, TUMBLE(order_time, INTERVAL '1' HOUR)
) WHERE row_num <= 5;

-- CDC merge: Upsert into warehouse
MERGE INTO orders_snowflake AS target
USING orders AS source
ON target.order_id = source.order_id
WHEN MATCHED THEN UPDATE SET amount = source.amount
WHEN NOT MATCHED THEN INSERT VALUES (
  source.order_id, source.customer_id, source.amount, source.region
);

PYTHON EXAMPLES (Dagster)

from dagster import asset, AssetIn, Definitions
import pandas as pd

@asset(
    name="orders_cleaned",
    group_name="transformations",
    io_manager_key="snowflake_io_manager",
)
def orders_cleaned(orders_raw: pd.DataFrame) -> pd.DataFrame:
    """Clean and deduplicate raw orders."""
    df = orders_raw.drop_duplicates(subset=["order_id"])
    df = df[df["amount"] > 0]
    return df

@asset(
    name="revenue_by_region",
    deps=[AssetIn("orders_cleaned"), AssetIn("customers")],
)
def revenue_by_region(
    orders_cleaned: pd.DataFrame,
    customers: pd.DataFrame,
) -> pd.DataFrame:
    """Aggregate revenue by region."""
    merged = orders_cleaned.merge(customers, on="customer_id")
    return merged.groupby("region").agg(
        total_revenue=("amount", "sum"),
        order_count=("order_id", "count"),
    ).reset_index()

# Partitioned asset
from dagster import DailyPartitionsDefinition

daily_partitions = DailyPartitionsDefinition(start_date="2026-01-01")

@asset(partition_def=daily_partitions)
def orders_partitioned(context):
    partition_date = context.partition_config["partition_date"]
    return pd.read_sql(
        f"SELECT * FROM orders WHERE date = '{partition_date}'", engine
    )
